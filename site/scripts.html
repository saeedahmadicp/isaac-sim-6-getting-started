"""Capture real Isaac Sim frames from the lesson 1.1 checkpoint scene.

First save the checkpoint, then run this helper from the Isaac Sim installation:
    ./python.sh learning_resource/scripts/m1_build_simple_robot.py --headless --test --save learning_resource/output/simple_robot.usd
    ./python.sh learning_resource/scripts/capture_first_robot.py --stage learning_resource/output/simple_robot.usd --output-dir learning_resource/screenshots/first_robot

The camera, lighting, and display colours are for readability. Geometry, joints,
and drives are loaded unchanged. This records simulated motion, not GUI actions.
"""

import argparse
import json
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("--stage", type=Path, required=True)
parser.add_argument("--output-dir", type=Path, required=True)
args, _ = parser.parse_known_args()
if not args.stage.is_file():
    parser.error(f"Stage does not exist: {args.stage}")
args.output_dir.mkdir(parents=True, exist_ok=True)

from isaacsim import SimulationApp

app = SimulationApp({"headless": True, "width": 1440, "height": 900,
                     "renderer": "RealTimePathTracing", "multi_gpu": False})

try:
    import carb
    import numpy as np
    import omni.replicator.core as rep
    import omni.timeline
    import omni.usd
    from PIL import Image
    from pxr import Gf, UsdGeom, UsdLux

    if not omni.usd.get_context().open_stage(str(args.stage.resolve())):
        raise RuntimeError("Could not open checkpoint stage")
    for _ in range(20):
        app.update()
    stage = omni.usd.get_context().get_stage()
    body = stage.GetPrimAtPath("/World/SimpleRobot/Cube")
    if not body.IsValid():
        raise RuntimeError("Expected the lesson 1.1 SimpleRobot checkpoint")
    xf = UsdGeom.Xformable(body)
    UsdGeom.Gprim(body).GetDisplayColorAttr().Set([Gf.Vec3f(0.12, 0.38, 0.65)])
    for prim in stage.Traverse():
        if prim.GetName().endswith("_Wheel"):
            UsdGeom.Gprim(prim).GetDisplayColorAttr().Set([Gf.Vec3f(0.16, 0.18, 0.22)])
    dome = UsdLux.DomeLight.Define(stage, "/World/CaptureDome")
    dome.GetIntensityAttr().Set(300.0)
    sun = UsdLux.DistantLight.Define(stage, "/World/defaultLight/DistantLight")
    sun.GetIntensityAttr().Set(1200.0)
    sun.GetAngleAttr().Set(2.0)
    UsdGeom.Xformable(sun).ClearXformOpOrder()
    UsdGeom.Xformable(sun).AddRotateXYZOp().Set(Gf.Vec3f(-45, 25, 0))
    settings = carb.settings.get_settings()
    settings.set("/rtx/post/tonemap/op", 4)
    settings.set("/rtx/post/tonemap/filmIso", 200.0)
    settings.set("/rtx/post/tonemap/enabled", True)
    camera = UsdGeom.Camera.Define(stage, "/World/CourseCamera")
    camera.CreateFocalLengthAttr(32.0)
    camera.CreateClippingRangeAttr(Gf.Vec2f(0.1, 100.0))
    camera_op = UsdGeom.Xformable(camera).AddTransformOp()

    def aim(eye, target, up=(0, 0, 1)):
        camera_op.Set(Gf.Matrix4d().SetLookAt(Gf.Vec3d(*eye), Gf.Vec3d(*target), Gf.Vec3d(*up)).GetInverse())

    aim((5.5, -7.0, 4.5), (0.5, 0, 0.3))
    rep.orchestrator.set_capture_on_play(False)
    product = rep.create.render_product(str(camera.GetPath()), (1440, 900))
    rgb = rep.AnnotatorRegistry.get_annotator("rgb")
    rgb.attach([product])
    for _ in range(60):
        app.update()
    frames = []

    def capture(name):
        rep.orchestrator.step(rt_subframes=16, delta_time=0.0, pause_timeline=True)
        pixels = np.asarray(rgb.get_data())[:, :, :3]
        if pixels.shape != (900, 1440, 3) or pixels.std() < 5 or not 10 < pixels.mean() < 220:
            raise RuntimeError(f"Invalid capture: shape={pixels.shape}, mean={pixels.mean()}, std={pixels.std()}")
        Image.fromarray(pixels).save(args.output_dir / name)
        frames.append({"file": name, "mean_rgb": round(float(pixels.mean()), 2),
                       "std_rgb": round(float(pixels.std()), 2),
                       "body_position": list(xf.ComputeLocalToWorldTransform(0).ExtractTranslation())})

    timeline = omni.timeline.get_timeline_interface()
    capture("assembled.png")
    timeline.play()
    for _ in range(180):
        app.update()
    timeline.pause()
    end_time = timeline.get_current_time()
    capture("driving.png")
    x = xf.ComputeLocalToWorldTransform(0).ExtractTranslation()[0]
    if not 0.5 < x < 1.5:
        raise RuntimeError(f"Unexpected driven displacement: {x}")
    aim((x, 0, 7), (x, 0, 0), (1, 0, 0))
    capture("wheel-layout.png")
    record = {"stage": str(args.stage), "frames": frames, "updates": 180,
              "timeline_seconds": end_time, "displacement_x_m": x,
              "presentation_changes": "Camera, lights, chassis/wheel display colours only; physics unchanged."}
    (args.output_dir / "capture.json").write_text(json.dumps(record, indent=2) + "\n")
    print(f"[capture] PASS: {len(frames)} frames, displacement={x:.3f} m, timeline={end_time:.3f} s")
    rgb.detach([product])
    product.destroy()
    timeline.stop()
finally:
    app.close()

"""Module 1 checkpoint: build the course's 4-wheeled "SimpleRobot" in Isaac Sim 6.1.

This reproduces lessons 1.1 and 1.2 of "Building Your First Robot in Isaac Sim" using the
same Kit commands the GUI menus call, so the result matches what you build by hand:

- Physics Scene + Ground Plane
- SimpleRobot (Xform) > Cube chassis (2 x 1 x 0.5 m) + 4 cylinder wheels (diameter 0.75 m)
- Rigid Body with Colliders Preset on chassis and wheels
- 4 revolute joints (axis Y, Local Rotation 1 = -90, 0, 0)
- Angular drives on the rear joints (damping 10000, target velocity 50 deg/s)
- Articulation Root on SimpleRobot (needed by the Differential Controller in lesson 1.2)

Usage (from the Isaac Sim root folder):
    ./python.sh learning_resource/scripts/m1_build_simple_robot.py            # GUI, keeps running
    ./python.sh learning_resource/scripts/m1_build_simple_robot.py --headless --test
    ./python.sh learning_resource/scripts/m1_build_simple_robot.py --save learning_resource/output/simple_robot.usd
"""

import argparse
import os

parser = argparse.ArgumentParser()
parser.add_argument("--headless", action="store_true", help="Run without a window")
parser.add_argument("--test", action="store_true", help="Simulate ~3 s, check the robot drove forward, then exit")
parser.add_argument("--save", default=None, help="Save the built stage to this .usd path")
args, _ = parser.parse_known_args()

from isaacsim import SimulationApp

simulation_app = SimulationApp({"headless": args.headless})

import omni.kit.commands
import omni.timeline
import omni.usd
from omni.physx.scripts import utils as physx_utils
from pxr import Gf, PhysxSchema, UsdGeom, UsdPhysics

omni.usd.get_context().new_stage()
simulation_app.update()
stage = omni.usd.get_context().get_stage()
UsdGeom.SetStageUpAxis(stage, UsdGeom.Tokens.z)
UsdGeom.SetStageMetersPerUnit(stage, 1.0)


def set_xform(path, translate=(0, 0, 0), rotate_xyz=(0, 0, 0), scale=(1, 1, 1)):
    """Set Translate / Orient / Scale, the same ops the Property > Transform panel edits.

    Rotation is given in XYZ euler degrees (what the panel shows) and stored as a quaternion.
    """
    prim = stage.GetPrimAtPath(path)
    xf = UsdGeom.Xformable(prim)
    xf.ClearXformOpOrder()
    rot = (
        Gf.Rotation(Gf.Vec3d(1, 0, 0), rotate_xyz[0])
        * Gf.Rotation(Gf.Vec3d(0, 1, 0), rotate_xyz[1])
        * Gf.Rotation(Gf.Vec3d(0, 0, 1), rotate_xyz[2])
    )
    xf.AddTranslateOp(UsdGeom.XformOp.PrecisionDouble).Set(Gf.Vec3d(*translate))
    xf.AddOrientOp(UsdGeom.XformOp.PrecisionDouble).Set(Gf.Quatd(rot.GetQuat()))
    xf.AddScaleOp(UsdGeom.XformOp.PrecisionDouble).Set(Gf.Vec3d(*scale))


def create_mesh(prim_type, path):
    """Same as Create > Mesh > <prim_type>."""
    omni.kit.commands.execute("CreateMeshPrimWithDefaultXform", prim_type=prim_type, prim_path=path)
    return stage.GetPrimAtPath(path)


# --- Lesson 1.1 Part 2: physics scene + ground plane --------------------------------------------
UsdPhysics.Scene.Define(stage, "/World/PhysicsScene")  # Create > Physics > Physics Scene
omni.kit.commands.execute(  # Create > Physics > Ground Plane
    "AddGroundPlaneCommand",
    stage=stage,
    planePath="/World/GroundPlane",
    axis="Z",
    size=25.0,
    position=Gf.Vec3f(0.0),
    color=Gf.Vec3f(0.5),
)
UsdGeom.Xform.Define(stage, "/World/defaultLight")
omni.kit.commands.execute("CreatePrim", prim_type="DistantLight", prim_path="/World/defaultLight/DistantLight")

# --- Lesson 1.1 Part 3: chassis + hierarchy -----------------------------------------------------
WHEEL_RADIUS = 0.375  # cylinder mesh is 1 m tall / 1 m diameter -> scale 0.75 gives 0.75 m diameter
UsdGeom.Xform.Define(stage, "/World/SimpleRobot")
set_xform("/World/SimpleRobot", translate=(0, 0, WHEEL_RADIUS))  # wheels rest on the ground
chassis = create_mesh("Cube", "/World/SimpleRobot/Cube")
set_xform(chassis.GetPath(), scale=(2, 1, 0.5))

# --- Lesson 1.1 Part 4: wheels ------------------------------------------------------------------
wheels = {
    "Front_Left_Wheel": (0.5, 0.75, 0),
    "Front_Right_Wheel": (0.5, -0.75, 0),
    "Rear_Left_Wheel": (-0.5, 0.75, 0),
    "Rear_Right_Wheel": (-0.5, -0.75, 0),
}
for name, pos in wheels.items():
    create_mesh("Cylinder", f"/World/SimpleRobot/{name}")
    set_xform(f"/World/SimpleRobot/{name}", translate=pos, rotate_xyz=(90, 0, 0), scale=(0.75, 0.75, 0.25))

# Right-click > Add > Physics > Rigid Body with Colliders Preset
for path in ["/World/SimpleRobot/Cube"] + [f"/World/SimpleRobot/{n}" for n in wheels]:
    physx_utils.setRigidBody(stage.GetPrimAtPath(path), "convexHull", False)

# --- Lesson 1.1 Part 5: revolute joints (select Cube first, then the wheel) ---------------------
local_rot1 = Gf.Rotation(Gf.Vec3d(1, 0, 0), -90).GetQuat()  # Local Rotation 1 = (-90, 0, 0)
joints = {}
for name in wheels:
    joint_name = name.replace("_Wheel", "_Joint")
    joint = physx_utils.createJoint(stage, "Revolute", chassis, stage.GetPrimAtPath(f"/World/SimpleRobot/{name}"))
    new_path = f"/World/SimpleRobot/{joint_name}"
    omni.kit.commands.execute("MovePrim", path_from=str(joint.GetPath()), path_to=new_path)
    joint = UsdPhysics.RevoluteJoint.Get(stage, new_path)
    joint.GetAxisAttr().Set("Y")
    joint.GetLocalRot0Attr().Set(Gf.Quatf(1, 0, 0, 0))
    joint.GetLocalRot1Attr().Set(Gf.Quatf(local_rot1))
    joints[joint_name] = joint.GetPrim()

# --- Lesson 1.1 Part 6: angular drive on the rear wheels ----------------------------------------
for joint_name in ["Rear_Left_Joint", "Rear_Right_Joint"]:
    drive = UsdPhysics.DriveAPI.Apply(joints[joint_name], "angular")  # Add > Physics > Angular Drive
    drive.CreateDampingAttr(10000.0)
    drive.CreateStiffnessAttr(0.0)
    drive.CreateTargetVelocityAttr(50.0)  # deg/s

# --- Lesson 1.2 Part 1: articulation root (required by the Differential Controller) -------------
robot_prim = stage.GetPrimAtPath("/World/SimpleRobot")
UsdPhysics.ArticulationRootAPI.Apply(robot_prim)
PhysxSchema.PhysxArticulationAPI.Apply(robot_prim)

if args.save:
    os.makedirs(os.path.dirname(os.path.abspath(args.save)), exist_ok=True)
    omni.usd.get_context().save_as_stage(os.path.abspath(args.save))
    print(f"[m1] saved stage to {os.path.abspath(args.save)}")

# --- Run ----------------------------------------------------------------------------------------
timeline = omni.timeline.get_timeline_interface()
timeline.play()

stage = omni.usd.get_context().get_stage()  # re-fetch: save_as_stage can swap the stage handle
chassis_xf = UsdGeom.Xformable(stage.GetPrimAtPath("/World/SimpleRobot/Cube"))
if args.test:
    for _ in range(180):  # ~3 s at 60 Hz
        simulation_app.update()
    x = chassis_xf.ComputeLocalToWorldTransform(0).ExtractTranslation()[0]
    print(f"[m1] chassis x after 3 s = {x:.3f} m")
    timeline.stop()
    print("[m1] PASS: robot drove forward" if x > 0.1 else "[m1] FAIL: robot did not drive forward")
    simulation_app.close()
else:
    while simulation_app.is_running():
        simulation_app.update()
    simulation_app.close()

"""Module 1.3/1.4 checkpoint: 2D lidar on SimpleRobot -> ROS 2 LaserScan -> RViz (Isaac Sim 6.1).

The course adds a *PhysX* rotating lidar and wires "Isaac Read Lidar Beams -> ROS2 Publish Laser Scan".
In 6.1 the PhysX lidar is deprecated (extsDeprecated/isaacsim.sensors.physx), so this uses the
replacement: an *RTX* lidar with the 2D config "Example_Rotary_2D", published through the same
graph that Tools > Robotics > ROS 2 OmniGraphs > RTX Lidar builds.

Expected: ~10 LaserScan msgs/s on /scan_lidar, frame sim_lidar, 3600 beams, and a forward beam of
2.50 m to an obstacle placed 2.50 m ahead.

Prerequisite: learning_resource/output/simple_robot.usd (run m1_build_simple_robot.py --save ... first,
or save your own robot there).

Usage (from the Isaac Sim root folder):
    ./python.sh learning_resource/scripts/m1_ros2_lidar.py
        then: ros2 topic echo /scan_lidar --once     /    rviz2  (Fixed Frame = sim_lidar, Add > By topic > /scan_lidar)
    ./python.sh learning_resource/scripts/m1_ros2_lidar.py --headless --test
"""

import argparse
import os

HERE = os.path.dirname(os.path.abspath(__file__))

parser = argparse.ArgumentParser()
parser.add_argument("--headless", action="store_true")
parser.add_argument("--test", action="store_true", help="Subscribe to /scan_lidar and check the obstacle is seen")
parser.add_argument("--usd", default=os.path.join(os.path.dirname(HERE), "output", "simple_robot.usd"))
args, _ = parser.parse_known_args()

from isaacsim import SimulationApp

simulation_app = SimulationApp({"headless": args.headless, "enable_motion_bvh": True})  # RTX lidar needs motion BVH

import omni.kit.app

omni.kit.app.get_app().get_extension_manager().set_extension_enabled_immediate("isaacsim.ros2.bridge", True)
simulation_app.update()  # let the ROS 2 bridge finish starting before touching the stage

import omni.kit.commands
import omni.timeline
import omni.usd
from isaacsim.ros2.nodes import Ros2RtxLidarGraphConfig, create_ros2_rtx_lidar_graph
from isaacsim.sensors.experimental.rtx import Lidar
from omni.physx.scripts import utils as physx_utils
from pxr import Gf, UsdGeom, UsdPhysics

omni.usd.get_context().open_stage(args.usd)
simulation_app.update()
stage = omni.usd.get_context().get_stage()

# Stop the rear-wheel drives from lesson 1.1 so the robot stands still for the scan check
for joint in ["Rear_Left_Joint", "Rear_Right_Joint"]:
    UsdPhysics.DriveAPI.Get(stage.GetPrimAtPath(f"/World/SimpleRobot/{joint}"), "angular").GetTargetVelocityAttr().Set(0.0)

# --- Lesson 1.3 (6.1 version): lidar on the chassis body -----------------------------------------
# The course mounts the lidar on Front_Sensor, its own rigid body fixed to the chassis with a Fixed Joint.
# On this 6.1 build an RTX lidar under such a fixed-jointed link published NOTHING (also under a
# scale-compensating Xform), while a lidar directly under the chassis body works - which is also how
# Nova Carter mounts its sensors. So: select Cube > Create > Sensors > RTX Lidar > NVIDIA > Example Rotary 2D.
# Its Translate is in Cube-local units (Cube is scaled 2 x 1 x 0.5): (0.45, 0, 1.0) = 0.9 m forward and
# 0.5 m above the chassis centre, i.e. above the wheel tops so the scan doesn't hit the robot itself.
lidar = Lidar.create(
    path="/World/SimpleRobot/Cube/Lidar", config="Example_Rotary_2D", tick_rate=10.0, translations=[[0.45, 0.0, 1.0]]
)
lidar_path = lidar.paths[0]

# An obstacle whose front face is 2.5 m in front of the lidar (lesson: add cubes/cones, Rigid Body with Colliders Preset)
omni.kit.commands.execute("CreateMeshPrimWithDefaultXform", prim_type="Cube", prim_path="/World/Obstacle")
# Create > Mesh prims carry translate/orient/scale ops; set the translate op directly (XformCommonAPI
# silently refuses prims that have an orient op).
stage.GetPrimAtPath("/World/Obstacle").GetAttribute("xformOp:translate").Set(Gf.Vec3d(3.9, 0.0, 0.5))
physx_utils.setRigidBody(stage.GetPrimAtPath("/World/Obstacle"), "convexHull", False)

# --- Lesson 1.4: Tools > Robotics > ROS 2 OmniGraphs > RTX Lidar (Laser Scan ticked) ------------
graph_path = create_ros2_rtx_lidar_graph(
    Ros2RtxLidarGraphConfig(
        graph_path="/World/ROS2_LidarStream",
        lidar_prim=lidar_path,
        frame_id="sim_lidar",
        publish_laser_scan=True,
        laser_scan_topic="scan_lidar",
    )
)
print(f"[m1] ROS 2 lidar graph: {graph_path}  lidar: {lidar_path}")

timeline = omni.timeline.get_timeline_interface()
timeline.play()

if args.test:
    import rclpy
    from rclpy.qos import qos_profile_sensor_data
    from sensor_msgs.msg import LaserScan

    rclpy.init()
    node = rclpy.create_node("m1_scan_check")
    scans = []
    node.create_subscription(LaserScan, "scan_lidar", lambda m: scans.append(m), qos_profile_sensor_data)
    import time

    t0, frames = time.time(), 0
    while (len(scans) < 3 or frames < 180) and frames < 1500:  # RTX sensors can take a while to warm up on the first run
        simulation_app.update()
        rclpy.spin_once(node, timeout_sec=0.0)
        frames += 1
    print(f"[m1] waited {frames} frames / {time.time() - t0:.1f} s")
    from isaacsim.core.experimental.prims import RigidPrim

    pos, _ = RigidPrim("/World/SimpleRobot/Cube").get_world_poses()
    print(f"[m1] chassis world position: {pos.numpy()[0].round(3)}")
    if scans:
        m = scans[-1]
        valid = [r for r in m.ranges if m.range_min < r < m.range_max]
        closest = min(valid) if valid else float("nan")
        import math

        def beam(angle_deg):
            i = int(round((math.radians(angle_deg) - m.angle_min) / m.angle_increment)) % len(m.ranges)
            return m.ranges[i]

        i_min = min(range(len(m.ranges)), key=lambda i: m.ranges[i] if m.range_min < m.ranges[i] < m.range_max else 1e9)
        print(f"[m1] forward beam = {beam(0):.2f} m (obstacle face expected at 2.50 m), "
